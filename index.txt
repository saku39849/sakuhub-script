      try {

        if (raw) {
          const parsed = JSON.parse(raw);
          return {
            games: Number(parsed.games) || 0,
            wins: Number(parsed.wins) || 0,
            losses: Number(parsed.losses) || 0,
            draws: Number(parsed.draws) || 0,
            streak: Number(parsed.streak) || 0,
          };
        }
      } catch (e) {
        // ストレージ非対応環境でも安全に続行
      }
      return { games: 0, wins: 0, losses: 0, draws: 0, streak: 0 };
    }

    /**
     * 指定列でコマが落ちる最下段の行番号を取得。満杯なら -1
     */
    function getAvailableRow(board, col) {
      if (col < 0 || col >= COLS) return -1;
      for (let r = ROWS - 1; r >= 0; r--) {
        if (board[r][col] === EMPTY) {
          return r;
        }
      }
      return -1;
    }

    /**
     * 盤面が満杯か判定
     */
    function isBoardFull(board) {
      for (let c = 0; c < COLS; c++) {
        if (board[0][c] === EMPTY) return false;
      }
      return true;
    }

    /**
     * 勝敗判定。勝利時は { winner, cells }、引き分けは { winner: 'DRAW', cells: [] }、続行中は null
     */
    function checkWinner(board) {
      const directions = [
        [0, 1],  // 横
        [1, 0],  // 縦
        [1, 1],  // 右下斜め
        [1, -1], // 左下斜め
      ];

      for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c < COLS; c++) {
          const player = board[r][c];
          if (player === EMPTY) continue;

          for (const [dr, dc] of directions) {
            const cells = [{ r, c }];
            let nr = r + dr;
            let nc = c + dc;
            while (
              nr >= 0 && nr < ROWS &&
              nc >= 0 && nc < COLS &&
              board[nr][nc] === player
            ) {
              cells.push({ r: nr, c: nc });
              if (cells.length === 4) {
                return { winner: player, cells };
              }
              nr += dr;
              nc += dc;
            }
          }
        }
      }

      if (isBoardFull(board)) {
        return { winner: 'DRAW', cells: [] };
      }
      return null;
    }

    /**
     * プレイヤーの列クリック処理
     */
    function handleHumanColumnClick(col) {
      if (state.gameOver || state.isAiThinking || state.currentTurn !== HUMAN) return;
      const row = getAvailableRow(state.board, col);
      if (row === -1) return;

      clearPreviewCol();
      stopTurnTimer();
      executeMove(row, col, HUMAN, false);
    }

    /**
     * コマを実際に盤面に落とし、状態遷移と次手スケジュールを行う
     */
    function executeMove(row, col, player, wasTimeout = false) {
      if (state.gameOver) return;
      if (state.board[row][col] !== EMPTY) return;

      state.history.push({
        board: cloneBoard(state.board),
        currentTurn: state.currentTurn,
        lastMove: state.lastMove ? { ...state.lastMove } : null,
        moveLog: state.moveLog.map(item => ({ ...item })),
        aiInsight: { ...state.lastAiInsight },
      });

      state.board[row][col] = player;
      state.lastMove = { r: row, c: col, player };
      const moveNumber = state.moveLog.length + 1;
      state.moveLog.push({
        moveNumber,
        player,
        col: col + 1,
        rowFromBottom: ROWS - row,
        wasTimeout,
      });

      const outcome = checkWinner(state.board);
      if (outcome) {
        state.gameOver = true;
        state.winner = outcome.winner;
        state.winningCells = outcome.cells;
        state.isAiThinking = false;
        recordMatchResult(outcome.winner);
        renderAll(true);
        return;
      }

      state.currentTurn = player === HUMAN ? AI : HUMAN;
      renderAll(true);

      if (state.currentTurn === AI) {
        scheduleAiTurn();
      } else {
        startTurnTimerIfNeeded();
      }
    }

    /**
     * 対局終了時の戦績反映
     */
    function recordMatchResult(winner) {
      state.stats.games += 1;
      if (winner === HUMAN) {
        state.stats.wins += 1;
        state.stats.streak = state.stats.streak > 0 ? state.stats.streak + 1 : 1;
      } else if (winner === AI) {
        state.stats.losses += 1;
        state.stats.streak = state.stats.streak < 0 ? state.stats.streak - 1 : -1;
      } else {
        state.stats.draws += 1;
        state.stats.streak = 0;
      }
      saveStats();
    }

    /**
     * AIの思考を非同期スケジュール
     */
    function scheduleAiTurn() {
      clearTimeout(state.aiTimeoutId);
      state.isAiThinking = true;
      renderStatusAndInsight();

      state.aiTimeoutId = setTimeout(() => {
        if (state.gameOver || state.currentTurn !== AI) {
          state.isAiThinking = false;
          renderStatusAndInsight();
          return;
        }

        const aiResult = computeBestMoveForAI(state.board, state.aiMode);
        state.lastAiInsight = {
          col: aiResult.col,
          score: aiResult.score,
          depth: aiResult.depth,
          nodes: aiResult.nodes,
          reasoning: aiResult.reasoning,
        };

        state.isAiThinking = false;
        const targetRow = getAvailableRow(state.board, aiResult.col);
        if (targetRow !== -1) {
          executeMove(targetRow, aiResult.col, AI, false);
        }
      }, 140);
    }

    /**
     * =========================================================================
     * 手加減なしAIエンジン（Minimax + Alpha-Beta + 反復深化 + 置換表）
     * =========================================================================
     */
    function computeBestMoveForAI(board, mode) {
      const validCols = MOVE_ORDER.filter(c => getAvailableRow(board, c) !== -1);
      if (validCols.length === 0) {
        return { col: 3, score: 0, depth: 0, nodes: 0, reasoning: '打てる手がありません。' };
      }

      // 1. 即時勝利手があれば即決
      for (const col of validCols) {
        const r = getAvailableRow(board, col);
        board[r][col] = AI;
        const win = checkWinner(board);
        board[r][col] = EMPTY;
        if (win && win.winner === AI) {
          return {
            col,
            score: WIN_SCORE,
            depth: 1,
            nodes: 1,
            reasoning: `${col + 1}列目に落として4目並べ完成。容赦なく仕留めます。`
          };
        }
      }

      // 2. 相手の即時勝利手（リーチ）があれば必ずブロック
      const immediateThreats = [];
      for (const col of validCols) {
        const r = getAvailableRow(board, col);
        board[r][col] = HUMAN;
        const win = checkWinner(board);
        board[r][col] = EMPTY;
        if (win && win.winner === HUMAN) {
          immediateThreats.push(col);
        }
      }
      if (immediateThreats.length > 0) {
        const blockCol = immediateThreats[0];
        return {
          col: blockCol,
          score: 50000,
          depth: 1,
          nodes: immediateThreats.length,
          reasoning: `あなたの${blockCol + 1}列目の勝ち筋を完全に遮断します。甘く見ないでください。`
        };
      }

      const emptyCount = countEmptyCells(board);
      const placedCount = ROWS * COLS - emptyCount;

      if (placedCount === 0) {
        return {
          col: 3,
          score: 120,
          depth: 1,
          nodes: 1,
          reasoning: '初手は中央4列目。盤面全体のライン数（21通り中13通り）を制圧する最善の起点です。'
        };
      }
      if (placedCount === 1 && getAvailableRow(board, 3) === ROWS - 1) {
        return {
          col: 3,
          score: 140,
          depth: 1,
          nodes: 1,
          reasoning: '中央4列目を即座に確保し、主導権を奪います。'
        };
      }

      let maxDepth = mode === 'kichiku' ? 8 : 7;
      if (emptyCount <= 14) maxDepth += 2;
      if (emptyCount <= 10) maxDepth += 2;
      maxDepth = Math.min(maxDepth, emptyCount);

      transpositionTable = new Map();
      let totalNodes = 0;
      let bestMove = validCols[0];
      let bestScore = -Infinity;
      let reachedDepth = 1;

      const startTime = performance.now();
      const timeBudgetMs = mode === 'kichiku' ? 1400 : 850;

      for (let depth = 1; depth <= maxDepth; depth++) {
        if (depth > 1 && performance.now() - startTime > timeBudgetMs) {
          break;
        }

        const searchResult = searchRoot(board, depth, totalNodes);
        totalNodes = searchResult.nodes;

        if (!searchResult.aborted) {
          bestMove = searchResult.col;
          bestScore = searchResult.score;
          reachedDepth = depth;
        }

        if (bestScore >= WIN_SCORE - 1000) {
          break;
        }
      }

      const reasoning = buildAiReasoning(board, bestMove, bestScore, mode);
      return {
        col: bestMove,
        score: bestScore,
        depth: reachedDepth,
        nodes: totalNodes,
        reasoning,
      };
    }

    /**
     * ルート階層での探索
     */
    function searchRoot(board, depth, initialNodes) {
      let nodes = initialNodes;
      let alpha = -Infinity;
      const beta = Infinity;
      let bestCol = MOVE_ORDER.find(c => getAvailableRow(board, c) !== -1);
      let bestScore = -Infinity;

      const orderedCols = orderMoves(board, AI);

      for (const col of orderedCols) {
        const row = getAvailableRow(board, col);
        if (row === -1) continue;

        board[row][col] = AI;
        nodes += 1;
        const score = -alphaBeta(board, depth - 1, -beta, -alpha, HUMAN, nodes);
        board[row][col] = EMPTY;

        if (score > bestScore) {
          bestScore = score;
          bestCol = col;
        }
        if (score > alpha) {
          alpha = score;
        }
      }

      return { col: bestCol, score: bestScore, nodes, aborted: false };
    }

    /**
     * ネガマックス形式の Alpha-Beta 探索 + 置換表
     */
    function alphaBeta(board, depth, alpha, beta, turnPlayer, nodesRef) {
      const alphaOrig = alpha;
      const hashKey = computeBoardHash(board, turnPlayer);
      const ttEntry = transpositionTable.get(hashKey);

      if (ttEntry && ttEntry.depth >= depth) {
        if (ttEntry.flag === TT_EXACT) return ttEntry.score;
        if (ttEntry.flag === TT_LOWERBOUND) alpha = Math.max(alpha, ttEntry.score);
        else if (ttEntry.flag === TT_UPPERBOUND) beta = Math.min(beta, ttEntry.score);
        if (alpha >= beta) return ttEntry.score;
      }

      const outcome = checkWinner(board);
      if (outcome) {
        if (outcome.winner === 'DRAW') return 0;
        const wonByCurrent = outcome.winner === turnPlayer;
        return wonByCurrent ? (WIN_SCORE - (10 - depth) * 1000) : -(WIN_SCORE - (10 - depth) * 1000);
      }

      if (depth === 0) {
        const rawEval = evaluateBoardForAI(board);
        return turnPlayer === AI ? rawEval : -rawEval;
      }

      const orderedCols = orderMoves(board, turnPlayer);
      let maxScore = -Infinity;
      const nextPlayer = turnPlayer === AI ? HUMAN : AI;

      for (const col of orderedCols) {
        const row = getAvailableRow(board, col);
        if (row === -1) continue;

        board[row][col] = turnPlayer;
        const score = -alphaBeta(board, depth - 1, -beta, -alpha, nextPlayer, nodesRef);
        board[row][col] = EMPTY;

        if (score > maxScore) {
          maxScore = score;
        }
        if (score > alpha) {
          alpha = score;
        }
        if (alpha >= beta) {
          break;
        }
      }

      let flag = TT_EXACT;
      if (maxScore <= alphaOrig) flag = TT_UPPERBOUND;
      else if (maxScore >= beta) flag = TT_LOWERBOUND;

      if (transpositionTable.size > 120000) {
        transpositionTable.clear();
      }
      transpositionTable.set(hashKey, { depth, score: maxScore, flag });

      return maxScore;
    }

    /**
     * 手の並び替え：即時勝ち・相手の即時勝ちブロック・中央列を優先
     */
    function orderMoves(board, player) {
      const opponent = player === AI ? HUMAN : AI;
      const scoredCols = [];

      for (const col of MOVE_ORDER) {
        const row = getAvailableRow(board, col);
        if (row === -1) continue;

        let priority = (3 - Math.abs(3 - col)) * 15;

        board[row][col] = player;
        if (checkWinner(board)?.winner === player) {
          priority += 100000;
        }
        board[row][col] = EMPTY;

        board[row][col] = opponent;
        if (checkWinner(board)?.winner === opponent) {
          priority += 50000;
        }
        board[row][col] = EMPTY;

        scoredCols.push({ col, priority });
      }

      scoredCols.sort((a, b) => b.priority - a.priority);
      return scoredCols.map(item => item.col);
    }

    /**
     * 盤面ハッシュ生成
     */
    function computeBoardHash(board, turnPlayer) {
      let str = turnPlayer === AI ? 'A:' : 'H:';
      for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c < COLS; c++) {
          str += board[r][c];
        }
      }
      return str;
    }

    /**
     * 空きマス数カウント
     */
    function countEmptyCells(board) {
      let count = 0;
      for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c < COLS; c++) {
          if (board[r][c] === EMPTY) count++;
        }
      }
      return count;
    }

    /**
     * AI視点の静的盤面評価関数
     */
    function evaluateBoardForAI(board) {
      let score = 0;

      const centerCol = Math.floor(COLS / 2);
      let aiCenterCount = 0;
      let humanCenterCount = 0;
      for (let r = 0; r < ROWS; r++) {
        if (board[r][centerCol] === AI) aiCenterCount++;
        else if (board[r][centerCol] === HUMAN) humanCenterCount++;
      }
      score += aiCenterCount * 18;
      score -= humanCenterCount * 16;

      // 横方向
      for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c <= COLS - 4; c++) {
          score += scoreWindow([board[r][c], board[r][c + 1], board[r][c + 2], board[r][c + 3]]);
        }
      }

      // 縦方向
      for (let c = 0; c < COLS; c++) {
        for (let r = 0; r <= ROWS - 4; r++) {
          score += scoreWindow([board[r][c], board[r + 1][c], board[r + 2][c], board[r + 3][c]]);
        }
      }

      // 右下斜め
      for (let r = 0; r <= ROWS - 4; r++) {
        for (let c = 0; c <= COLS - 4; c++) {
          score += scoreWindow([board[r][c], board[r + 1][c + 1], board[r + 2][c + 2], board[r + 3][c + 3]]);
        }
      }

      // 左下斜め
      for (let r = 3; r < ROWS; r++) {
        for (let c = 0; c <= COLS - 4; c++) {
          score += scoreWindow([board[r][c], board[r - 1][c + 1], board[r - 2][c + 2], board[r - 3][c + 3]]);
        }
      }

      return score;
    }

    /**
     * 4マスウィンドウの評価スコア計算
     */
    function scoreWindow(window) {
      let aiCount = 0;
      let humanCount = 0;
      let emptyCount = 0;

      for (let i = 0; i < 4; i++) {
        if (window[i] === AI) aiCount++;
        else if (window[i] === HUMAN) humanCount++;
        else emptyCount++;
      }

      if (aiCount > 0 && humanCount > 0) return 0;
      if (aiCount === 4) return 100000;
      if (humanCount === 4) return -100000;

      let delta = 0;
      if (aiCount === 3 && emptyCount === 1) {
        delta += 140;
      } else if (aiCount === 2 && emptyCount === 2) {
        delta += 22;
      }

      if (humanCount === 3 && emptyCount === 1) {
        delta -= 175;
      } else if (humanCount === 2 && emptyCount === 2) {
        delta -= 20;
      }

      return delta;
    }

    /**
     * AIの狙い・解説文を生成
     */
    function buildAiReasoning(board, col, score, mode) {
      const modeTag = mode === 'kichiku' ? '【鬼畜深読み】' : '【本気モード】';
      if (score >= 900000) {
        return `${modeTag} ${col + 1}列目への着手から必勝ライン（強制勝利）が確定しています。`;
      }
      if (score <= -900000) {
        return `${modeTag} 厳しい局面ですが、${col + 1}列目で粘りつつ相手のミスを待ちます。`;
      }
      if (col === 3) {
        return `${modeTag} 中央4列目を選択。縦・横・斜めの複数ラインを同時に伸ばし、次の攻め手を広げます。`;
      }
      if (score > 120) {
        return `${modeTag} ${col + 1}列目に打って有利な形勢を拡大。相手の防御が追いつかない二重の脅威を狙います。`;
      }
      return `${modeTag} ${col + 1}列目を選択。あなたの連を作らせず、中央付近の主導権とライン効率を最適化しました。`;
    }

    /**
     * 一手戻す（Undo）処理
     */
    function handleUndoMove() {
      if (state.isAiThinking || state.history.length === 0) return;

      stopTurnTimer();
      clearTimeout(state.aiTimeoutId);

      if (state.gameOver && state.winner) {
        rollbackLastStat();
      }

      let targetSnapshot = state.history.pop();
      while (
        targetSnapshot &&
        targetSnapshot.currentTurn !== HUMAN &&
        state.history.length > 0
      ) {
        targetSnapshot = state.history.pop();
      }

      if (!targetSnapshot) return;

      state.board = cloneBoard(targetSnapshot.board);
      state.currentTurn = targetSnapshot.currentTurn;
      state.lastMove = targetSnapshot.lastMove ? { ...targetSnapshot.lastMove } : null;
      state.moveLog = targetSnapshot.moveLog.map(item => ({ ...item }));
      state.lastAiInsight = { ...targetSnapshot.aiInsight };
      state.gameOver = false;
      state.winner = null;
      state.winningCells = [];
      state.isAiThinking = false;

      renderAll(false);
      if (state.currentTurn === AI) {
        scheduleAiTurn();
      } else {
        startTurnTimerIfNeeded();
      }
    }

    /**
     * Undo時に直近1局の戦績を巻き戻す
     */
    function rollbackLastStat() {
      if (state.stats.games > 0) {
        state.stats.games -= 1;
      }
      if (state.winner === HUMAN && state.stats.wins > 0) {
        state.stats.wins -= 1;
      } else if (state.winner === AI && state.stats.losses > 0) {
        state.stats.losses -= 1;
      } else if (state.winner === 'DRAW' && state.stats.draws > 0) {
        state.stats.draws -= 1;
      }
      state.stats.streak = 0;
      saveStats();
    }

    /**
     * 盤面をリセットして新しい対局を開始
     */
    function resetMatch() {
      stopTurnTimer();
      clearTimeout(state.aiTimeoutId);
      clearPreviewCol();

      state.board = createEmptyBoard();
      state.currentTurn = state.humanStarts ? HUMAN : AI;
      state.gameOver = false;
      state.winner = null;
      state.winningCells = [];
      state.lastMove = null;
      state.history = [];
      state.moveLog = [];
      state.isAiThinking = false;
      state.lastAiInsight = {
        col: null,
        score: 0,
        depth: 0,
        nodes: 0,
        reasoning: '中央列（4列目）を軸に、縦・横・斜めの同時脅威（ダブルスレット）と相手のリーチ阻止を最優先で計算します。'
      };

      renderAll(false);

      if (state.currentTurn === AI) {
        scheduleAiTurn();
      } else {
        startTurnTimerIfNeeded();
      }
    }

    /**
     * 15秒ターンタイマー開始
     */
    function startTurnTimerIfNeeded() {
      stopTurnTimer();
      if (!state.timerEnabled || state.gameOver || state.isAiThinking || state.currentTurn !== HUMAN) {
        updateTimerBarUI(15.0);
        return;
      }

      state.timeLeft = 15.0;
      updateTimerBarUI(state.timeLeft);

      const tickMs = 100;
      state.timerId = setInterval(() => {
        if (state.gameOver || state.currentTurn !== HUMAN || state.isAiThinking) {
          stopTurnTimer();
          return;
        }

        state.timeLeft = Math.max(0, +(state.timeLeft - tickMs / 1000).toFixed(1));
        updateTimerBarUI(state.timeLeft);

        if (state.timeLeft <= 0) {
          stopTurnTimer();
          const openCols = [];
          for (let c = 0; c < COLS; c++) {
            if (getAvailableRow(state.board, c) !== -1) openCols.push(c);
          }
          if (openCols.length > 0) {
            const randCol = openCols[Math.floor(Math.random() * openCols.length)];
            const targetRow = getAvailableRow(state.board, randCol);
            clearPreviewCol();
            executeMove(targetRow, randCol, HUMAN, true);
          }
        }
      }, tickMs);
    }

    /**
     * 15秒ターンタイマー停止
     */
    function stopTurnTimer() {
      if (state.timerId) {
        clearInterval(state.timerId);
        state.timerId = null;
      }
    }

    /**
     * タイマーゲージUI更新
     */
    function updateTimerBarUI(seconds) {
      dom.timerSecondsText.textContent = `${seconds.toFixed(1)}s`;
      const pct = Math.max(0, Math.min(100, (seconds / 15) * 100));
      dom.timerProgressBar.style.width = `${pct}%`;

      if (seconds <= 4.0) {
        dom.timerSecondsText.className = 'font-mono text-sm text-red-400 animate-pulse';
      } else if (seconds <= 8.0) {
        dom.timerSecondsText.className = 'font-mono text-sm text-amber-400';
      } else {
        dom.timerSecondsText.className = 'font-mono text-sm text-emerald-400';
      }
    }

    /**
     * 画面全体の描画更新
     */
    function renderAll(animateLastMove = false) {
      renderBoardCells(animateLastMove);
      renderStatusAndInsight();
      renderStats();
      renderMoveHistory();

      dom.undoBtn.disabled = state.isAiThinking || state.history.length === 0;
    }

    /**
     * 42マスのコマ描画を更新
     */
    function renderBoardCells(animateLastMove) {
      const cells = dom.boardGrid.children;
      const winSet = new Set(state.winningCells.map(pt => `${pt.r},${pt.c}`));

      for (let r = 0; r < ROWS; r++) {
        for (let c = 0; c < COLS; c++) {
          const idx = r * COLS + c;
          const slot = cells[idx];
          const val = state.board[r][c];
          slot.innerHTML = '';

          if (val !== EMPTY) {
            const disc = document.createElement('div');
            disc.className = `w-full h-full rounded-full relative ${val === HUMAN ? 'disc-red' : 'disc-yellow'}`;

            if (state.lastMove && state.lastMove.r === r && state.lastMove.c === c) {
              const marker = document.createElement('span');
              marker.className = 'w-2 h-2 sm:w-2.5 sm:h-2.5 rounded-full bg-white/85 shadow pointer-events-none block absolute top-1/2 left-1/2 -translate-x-1/2 -translate-y-1/2';
              disc.appendChild(marker);

              if (animateLastMove) {
                const dropDistance = (r + 1) * 56;
                const duration = Math.min(460, 220 + r * 38);
                disc.style.setProperty('--drop-start', `-${dropDistance}px`);
                disc.style.setProperty('--drop-duration', `${duration}ms`);
                disc.classList.add('disc-drop');
              }
            }

            if (winSet.has(`${r},${c}`)) {
              disc.classList.add('disc-win');
            }

            slot.appendChild(disc);
          }
        }
      }
    }

    /**
     * 手番表示・ステータス・AI思考インサイトの更新
     */
    function renderStatusAndInsight() {
      const currentMoveNum = state.moveLog.length + 1;
      dom.turnCountLabel.textContent = state.gameOver
        ? `全 ${state.moveLog.length} 手で決着`
        : `第 ${currentMoveNum} 手`;

      if (state.gameOver) {
        if (state.winner === HUMAN) {
          dom.statusHeadline.textContent = 'あなたの勝利！';
          dom.statusHeadline.className = 'text-xs sm:text-sm font-black text-emerald-400 leading-tight mt-0.5';
        } else if (state.winner === AI) {
          dom.statusHeadline.textContent = 'AI の勝利';
          dom.statusHeadline.className = 'text-xs sm:text-sm font-black text-red-400 leading-tight mt-0.5';
        } else {
          dom.statusHeadline.textContent = '引き分け（満杯）';
          dom.statusHeadline.className = 'text-xs sm:text-sm font-black text-amber-300 leading-tight mt-0.5';
        }
      } else if (state.isAiThinking) {
        dom.statusHeadline.textContent = 'AI が深読み中...';
        dom.statusHeadline.className = 'text-xs sm:text-sm font-black text-amber-300 leading-tight mt-0.5';
      } else if (state.currentTurn === HUMAN) {
        dom.statusHeadline.textContent = 'あなたの番（列をクリック）';
        dom.statusHeadline.className = 'text-xs sm:text-sm font-black text-red-400 leading-tight mt-0.5';
      } else {
        dom.statusHeadline.textContent = 'AI の手番';
        dom.statusHeadline.className = 'text-xs sm:text-sm font-black text-amber-300 leading-tight mt-0.5';
      }

      if (!state.gameOver && state.currentTurn === HUMAN && !state.isAiThinking) {
        dom.youCard.className = 'flex items-center gap-2.5 p-2 rounded-xl bg-slate-950 border-2 border-red-500 shadow-redGlow transition';
        dom.youTurnBadge.textContent = 'あなたの番';
        dom.youTurnBadge.className = 'text-[11px] font-extrabold text-red-400';

        dom.aiCard.className = 'flex items-center justify-end gap-2.5 p-2 rounded-xl bg-slate-950/70 border border-slate-800 transition text-right opacity-75';
        dom.aiTurnBadge.textContent = '待機中';
        dom.aiTurnBadge.className = 'text-[11px] font-bold text-slate-500';
      } else if (!state.gameOver && (state.currentTurn === AI || state.isAiThinking)) {
        dom.aiCard.className = 'flex items-center justify-end gap-2.5 p-2 rounded-xl bg-slate-950 border-2 border-amber-400 shadow-yellowGlow transition text-right';
        dom.aiTurnBadge.textContent = state.isAiThinking ? '思考中...' : 'AIの番';
        dom.aiTurnBadge.className = 'text-[11px] font-extrabold text-amber-300';

        dom.youCard.className = 'flex items-center gap-2.5 p-2 rounded-xl bg-slate-950/70 border border-slate-800 transition opacity-75';
        dom.youTurnBadge.textContent = '待機中';
        dom.youTurnBadge.className = 'text-[11px] font-bold text-slate-500';
      } else {
        dom.youCard.className = 'flex items-center gap-2.5 p-2 rounded-xl bg-slate-950/80 border border-slate-800 transition';
        dom.aiCard.className = 'flex items-center justify-end gap-2.5 p-2 rounded-xl bg-slate-950/80 border border-slate-800 transition text-right';
        dom.youTurnBadge.textContent = '対局終了';
        dom.aiTurnBadge.textContent = '対局終了';
      }

      dom.aiInsightPanel.classList.toggle('hidden', !state.showEval);
      if (state.showEval) {
        dom.aiThinkingPulse.className = state.isAiThinking
          ? 'w-2.5 h-2.5 rounded-full bg-amber-400 animate-ping'
          : 'w-2.5 h-2.5 rounded-full bg-emerald-400';

        dom.aiDepthBadge.textContent = state.lastAiInsight.depth > 0
          ? `探索深度 ${state.lastAiInsight.depth}`
          : '探索深度 --';

        dom.aiStateText.textContent = state.isAiThinking
          ? '最善手を計算中（手加減なし）'
          : state.gameOver
            ? '対局完了'
            : 'あなたの着手待ち';

        dom.aiLastMoveText.textContent = state.lastAiInsight.col !== null
          ? `${state.lastAiInsight.col + 1} 列目`
          : '未着手';

        const score = state.lastAiInsight.score;
        let evalLabel = `${score > 0 ? '+' : ''}${score}`;
        let evalColor = 'text-slate-200';
        if (score >= 900000) {
          evalLabel = 'AI 必勝 (+∞)';
          evalColor = 'text-amber-300';
        } else if (score <= -900000) {
          evalLabel = 'AI 劣勢 (-∞)';
          evalColor = 'text-emerald-400';
        } else if (score > 80) {
          evalLabel = `${score > 0 ? '+' : ''}${score} (AI優勢)`;
          evalColor = 'text-amber-300';
        } else if (score < -80) {
          evalLabel = `${score} (あなた優勢)`;
          evalColor = 'text-emerald-400';
        } else {
          evalLabel = `${score > 0 ? '+' : ''}${score} (互角)`;
          evalColor = 'text-sky-300';
        }
        dom.aiEvalScoreText.textContent = evalLabel;
        dom.aiEvalScoreText.className = `font-mono font-extrabold ${evalColor}`;
        dom.aiReasoningText.textContent = state.lastAiInsight.reasoning;
      }
    }

    /**
     * 戦績表示の更新
     */
    function renderStats() {
      dom.statGames.textContent = String(state.stats.games);
      dom.statWins.textContent = String(state.stats.wins);
      dom.statLosses.textContent = String(state.stats.losses);
      dom.statDraws.textContent = String(state.stats.draws);

      if (state.stats.streak > 0) {
        dom.streakBadge.textContent = `${state.stats.streak} 連勝中`;
        dom.streakBadge.className = 'px-2.5 py-0.5 rounded-full font-extrabold bg-emerald-500/20 text-emerald-300 border border-emerald-500/30';
      } else if (state.stats.streak < 0) {
        dom.streakBadge.textContent = `${Math.abs(state.stats.streak)} 連敗中`;
        dom.streakBadge.className = 'px-2.5 py-0.5 rounded-full font-extrabold bg-red-500/20 text-red-300 border border-red-500/30';
      } else {
        dom.streakBadge.textContent = '記録なし';
        dom.streakBadge.className = 'px-2.5 py-0.5 rounded-full font-extrabold bg-slate-800 text-slate-300';
      }
    }

    /**
     * 着手ログの更新
     */
    function renderMoveHistory() {
      dom.moveHistoryCount.textContent = `${state.moveLog.length} 手`;

      if (state.moveLog.length === 0) {
        dom.moveHistoryList.innerHTML = '<div class="text-slate-500 text-center py-4">まだ手が指されていません</div>';
        return;
      }

      const itemsHtml = state.moveLog
        .slice()
        .reverse()
        .map(entry => {
          const isHuman = entry.player === HUMAN;
          const badgeColor = isHuman
            ? 'bg-red-500/20 text-red-300 border-red-500/30'
            : 'bg-amber-400/20 text-amber-300 border-amber-400/30';
          const actorName = isHuman ? 'あなた (赤)' : 'AI (黄)';
          const timeoutTag = entry.wasTimeout ? ' <span class="text-red-400 font-bold">[時間切れ]</span>' : '';

          return `
            <div class="flex items-center justify-between bg-slate-950/90 border border-slate-800/80 rounded-lg px-2.5 py-1.5">
              <div class="flex items-center gap-2">
                <span class="font-mono text-slate-500 w-7">#${entry.moveNumber}</span>
                <span class="px-2 py-0.5 rounded border text-[11px] font-bold ${badgeColor}">${actorName}</span>
              </div>
              <div class="font-mono text-slate-200 font-bold">
                ${entry.col}列目（下から${entry.rowFromBottom}段）${timeoutTag}
              </div>
            </div>
          `;
        })
        .join('');

      dom.moveHistoryList.innerHTML = itemsHtml;
    }

    /**
     * イベントリスナー登録
     */
    function bindEvents() {
      dom.resetBtn.addEventListener('click', resetMatch);
      dom.undoBtn.addEventListener('click', handleUndoMove);

      dom.modeSeriousBtn.addEventListener('click', () => {
        state.aiMode = 'serious';
        dom.modeSeriousBtn.className = 'px-3 py-1.5 rounded-lg text-xs font-bold transition bg-red-600 text-white shadow cursor-pointer';
        dom.modeKichikuBtn.className = 'px-3 py-1.5 rounded-lg text-xs font-bold transition text-slate-400 hover:text-white cursor-pointer';
        renderStatusAndInsight();
      });

      dom.modeKichikuBtn.addEventListener('click', () => {
        state.aiMode = 'kichiku';
        dom.modeKichikuBtn.className = 'px-3 py-1.5 rounded-lg text-xs font-bold transition bg-gradient-to-r from-red-600 to-amber-500 text-white shadow cursor-pointer';
        dom.modeSeriousBtn.className = 'px-3 py-1.5 rounded-lg text-xs font-bold transition text-slate-400 hover:text-white cursor-pointer';
        renderStatusAndInsight();
      });

      dom.timerToggleBtn.addEventListener('click', () => {
        state.timerEnabled = !state.timerEnabled;
        if (state.timerEnabled) {
          dom.timerStatusDot.className = 'w-2 h-2 rounded-full bg-emerald-400 animate-pulse';
          dom.timerToggleLabel.textContent = 'ON（15秒制限）';
          dom.timerBarContainer.classList.remove('hidden');
          startTurnTimerIfNeeded();
        } else {
          dom.timerStatusDot.className = 'w-2 h-2 rounded-full bg-slate-500';
          dom.timerToggleLabel.textContent = 'OFF（無制限特訓）';
          dom.timerBarContainer.classList.add('hidden');
          stopTurnTimer();
        }
      });

      dom.starterToggleBtn.addEventListener('click', () => {
        state.humanStarts = !state.humanStarts;
        dom.starterToggleBtn.textContent = state.humanStarts
          ? 'あなた先攻（赤）'
          : 'AI先攻（黄・容赦なし）';
        resetMatch();
      });

      dom.evalVisibilityBtn.addEventListener('click', () => {
        state.showEval = !state.showEval;
        dom.evalVisibilityBtn.textContent = state.showEval ? '解析表示：ON' : '解析表示：OFF';
        renderStatusAndInsight();
      });

      dom.resetStatsBtn.addEventListener('click', () => {
        state.stats = { games: 0, wins: 0, losses: 0, draws: 0, streak: 0 };
        saveStats();
        renderStats();
      });

      document.addEventListener('keydown', (e) => {
        if (e.target && (e.target.tagName === 'INPUT' || e.target.tagName === 'TEXTAREA')) return;

        if (e.key >= '1' && e.key <= '7') {
          const colIdx = Number(e.key) - 1;
          handleHumanColumnClick(colIdx);
        } else if (e.key.toLowerCase() === 'z') {
          handleUndoMove();
        } else if (e.key.toLowerCase() === 'r') {
          resetMatch();
        }
      });
    }

    // アプリ初期化
    window.addEventListener('DOMContentLoaded', () => {
      initBoardDOM();
      bindEvents();
      renderAll(false);
      startTurnTimerIfNeeded();
    });
  </script>
